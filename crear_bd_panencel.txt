create table public.profiles (
  id uuid primary key references auth.users (id) on delete cascade,
  full_name text not null default 'Cliente',
  role text not null default 'cliente'
    check (role in ('admin', 'cliente')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.categories (
  id uuid primary key default gen_random_uuid(),
  name text not null unique,
  description text not null default '',
  is_active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.products (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  description text not null default '',
  price numeric(12, 2) not null check (price >= 0),
  emoji text not null default '',
  is_available boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.category_products (
  category_id uuid not null references public.categories (id) on delete cascade,
  product_id  uuid not null references public.products  (id) on delete cascade,
  created_at timestamptz not null default now(),
  primary key (category_id, product_id)
);

create index products_name_idx on public.products (name);
create index category_products_product_id_idx
  on public.category_products (product_id);

create table public.orders (
  id uuid primary key default gen_random_uuid(),
  code text not null unique default (
    'PN-' || upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 10))
  ),
  customer_id uuid not null references public.profiles (id) on delete restrict,
  fulfillment_status text not null default 'pending'
    check (fulfillment_status in ('pending', 'preparing', 'ready', 'completed', 'cancelled')),
  payment_status text not null default 'pending'
    check (payment_status in ('pending', 'paid', 'failed', 'refunded')),
  payment_method text not null
    check (payment_method in ('efectivo', 'tarjeta')),
  -- B4: admite 0 (productos con price = 0 son válidos por products.price >= 0)
  total_amount numeric(12, 2) not null check (total_amount >= 0),
  paid_at timestamptz,
  paid_validated_by uuid references public.profiles (id) on delete set null,
  payment_reference text unique,
  completed_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  check (
    (payment_status in ('paid', 'refunded') and paid_at is not null)
    or (payment_status in ('pending', 'failed') and paid_at is null)
  )
);

create index orders_customer_id_created_at_idx
  on public.orders (customer_id, created_at desc);
create index orders_payment_status_created_at_idx
  on public.orders (payment_status, created_at desc);
create index orders_fulfillment_status_created_at_idx
  on public.orders (fulfillment_status, created_at desc);

create table public.order_items (
  id uuid primary key default gen_random_uuid(),
  order_id uuid not null references public.orders (id) on delete cascade,
  product_id uuid references public.products (id) on delete set null,
  product_name text not null,
  unit_price numeric(12, 2) not null check (unit_price >= 0),
  -- B10: tope defensivo superior
  quantity integer not null check (quantity > 0 and quantity <= 999),
  line_total numeric(14, 2)
    generated always as (unit_price * quantity) stored,
  created_at timestamptz not null default now()
);

create index order_items_order_id_idx on public.order_items (order_id);
create index order_items_product_id_idx on public.order_items (product_id);

-- Crea el perfil al registrarse en Supabase Auth. El rol nunca viene del cliente.
create or replace function public.handle_new_auth_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles (id, full_name)
  values (
    new.id,
    coalesce(nullif(trim(new.raw_user_meta_data ->> 'full_name'), ''), 'Cliente')
  )
  on conflict (id) do nothing;
  return new;
end;
$$;

create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_auth_user();

-- is_admin(): consulta segura del rol administrativo para RLS y vistas.
create or replace function public.is_admin()
returns boolean
language sql
stable
security definer
set search_path = ''
as $$
  select exists (
    select 1
    from public.profiles as profile
    where profile.id = (select auth.uid())
      and profile.role = 'admin'
  );
$$;

-- ---------------------------------------------------------------------------
-- create_order: B6 (regex antes de cast), B7 (agrega 1 vez), B8 (retorna jsonb)
-- ---------------------------------------------------------------------------
create or replace function public.create_order(
  p_items jsonb,
  p_payment_method text
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_customer_id uuid := auth.uid();
  v_order_id uuid;
  v_order_code text;
  v_total numeric(12, 2);
  v_product_ids uuid[];
  v_quantities int[];
begin
  if v_customer_id is null then
    raise exception 'Se requiere iniciar sesion para crear un pedido.'
      using errcode = '42501';
  end if;

  if p_payment_method is null
    or p_payment_method not in ('efectivo', 'tarjeta') then
    raise exception 'Metodo de pago no valido.'
      using errcode = '22023';
  end if;

  if jsonb_typeof(p_items) is distinct from 'array' then
    raise exception 'Los productos deben enviarse como un arreglo JSON.'
      using errcode = '22023';
  end if;

  if jsonb_array_length(p_items) = 0 then
    raise exception 'El pedido debe contener al menos un producto.'
      using errcode = '22023';
  end if;

  -- B6: validación estructural ANTES de castear (evita errores crudos de uuid/int)
  if exists (
    select 1
    from jsonb_array_elements(p_items) as elem
    where jsonb_typeof(elem) is distinct from 'object'
      or (elem ->> 'product_id') is null
      or (elem ->> 'product_id') !~*
          '^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$'
      or (elem ->> 'quantity') is null
      or (elem ->> 'quantity') !~ '^[0-9]+$'
  ) then
    raise exception 'Cada producto debe tener product_id (UUID) y quantity (entero).'
      using errcode = '22023';
  end if;

  -- B7: agregación única reutilizable
  select array_agg(agg.product_id), array_agg(agg.quantity)
  into v_product_ids, v_quantities
  from (
    select
      (elem ->> 'product_id')::uuid as product_id,
      sum((elem ->> 'quantity')::int)::int as quantity
    from jsonb_array_elements(p_items) as elem
    group by 1
  ) as agg;

  if exists (
    select 1
    from unnest(v_product_ids, v_quantities) as item(product_id, quantity)
    where item.quantity < 1 or item.quantity > 99
  ) then
    raise exception 'Cada producto debe tener cantidad entre 1 y 99.'
      using errcode = '22023';
  end if;

  -- Bloqueo estable de productos (evita deadlocks, congela precio/disponibilidad)
  perform product.id
  from public.products as product
  join unnest(v_product_ids) as req(product_id) on req.product_id = product.id
  order by product.id
  for share of product;

  if exists (
    select 1
    from unnest(v_product_ids) as req(product_id)
    left join public.products as product on product.id = req.product_id
    where product.id is null or not product.is_available
  ) then
    raise exception 'Uno o mas productos no existen o no estan disponibles.'
      using errcode = '22023';
  end if;

  select sum(product.price * item.quantity)::numeric(12, 2)
  into v_total
  from unnest(v_product_ids, v_quantities) as item(product_id, quantity)
  join public.products as product on product.id = item.product_id;

  insert into public.orders (customer_id, payment_method, total_amount)
  values (v_customer_id, p_payment_method, v_total)
  returning id, code into v_order_id, v_order_code;

  insert into public.order_items (order_id, product_id, product_name, unit_price, quantity)
  select v_order_id, product.id, product.name, product.price, item.quantity
  from unnest(v_product_ids, v_quantities) as item(product_id, quantity)
  join public.products as product on product.id = item.product_id;

  -- B8: devuelve el code para armar el QR sin SELECT extra
  return jsonb_build_object(
    'id', v_order_id,
    'code', v_order_code,
    'total_amount', v_total
  );
end;
$$;

-- ---------------------------------------------------------------------------
-- admin_get_order_by_qr: B9 (regex estricto sobre el payload)
-- ---------------------------------------------------------------------------
create or replace function public.admin_get_order_by_qr(p_qr_payload text)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_code text;
  v_order jsonb;
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Solo un administrador puede consultar pedidos por QR.'
      using errcode = '42501';
  end if;

  if p_qr_payload is null
    or p_qr_payload !~ '^PANENCEL:ORDER:[A-Z0-9-]{1,64}$' then
    raise exception 'El codigo QR no corresponde a un pedido PANENCEL.'
      using errcode = '22023';
  end if;

  v_code := substring(p_qr_payload from length('PANENCEL:ORDER:') + 1);

  select jsonb_build_object(
    'code', orders.code,
    'customer_name', profile.full_name,
    'payment_method', orders.payment_method,
    'payment_status', orders.payment_status,
    'fulfillment_status', orders.fulfillment_status,
    'total_amount', orders.total_amount,
    'created_at', orders.created_at,
    'items', coalesce(
      (
        select jsonb_agg(
          jsonb_build_object(
            'product_id', items.product_id,
            'name', items.product_name,
            'unit_price', items.unit_price,
            'quantity', items.quantity,
            'line_total', items.line_total
          )
          order by items.created_at, items.id
        )
        from public.order_items as items
        where items.order_id = orders.id
      ),
      '[]'::jsonb
    )
  )
  into v_order
  from public.orders as orders
  join public.profiles as profile on profile.id = orders.customer_id
  where orders.code = v_code;

  if v_order is null then
    raise exception 'No se encontro el pedido del codigo QR.'
      using errcode = 'P0002';
  end if;

  return v_order;
end;
$$;

-- ---------------------------------------------------------------------------
-- admin_confirm_cash_payment: B3 (idempotente)
-- ---------------------------------------------------------------------------
create or replace function public.admin_confirm_cash_payment(p_order_code text)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Solo un administrador puede confirmar pagos en efectivo.'
      using errcode = '42501';
  end if;

  select * into v_order
  from public.orders
  where code = p_order_code
    and payment_method = 'efectivo';

  if not found then
    raise exception 'El pedido no existe o no admite pago en efectivo.'
      using errcode = '22023';
  end if;

  if v_order.fulfillment_status = 'cancelled' then
    raise exception 'El pedido fue cancelado.' using errcode = '22023';
  end if;

  if v_order.payment_status = 'paid' then
    return jsonb_build_object(
      'code', v_order.code,
      'payment_status', v_order.payment_status,
      'paid_at', v_order.paid_at,
      'paid_validated_by', v_order.paid_validated_by
    );
  end if;

  update public.orders
  set payment_status = 'paid',
      paid_at = now(),
      paid_validated_by = auth.uid()
  where id = v_order.id
  returning * into v_order;

  return jsonb_build_object(
    'code', v_order.code,
    'payment_status', v_order.payment_status,
    'paid_at', v_order.paid_at,
    'paid_validated_by', v_order.paid_validated_by
  );
end;
$$;

-- ---------------------------------------------------------------------------
-- confirm_card_order_payment: B2 (idempotente ante reintentos de webhook)
-- Solo invocable por service_role.
-- ---------------------------------------------------------------------------
create or replace function public.confirm_card_order_payment(
  p_order_code text,
  p_payment_reference text
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_reference text := trim(coalesce(p_payment_reference, ''));
begin
  if v_reference = '' then
    raise exception 'Se requiere la referencia confirmada del proveedor de pago.'
      using errcode = '22023';
  end if;

  select * into v_order
  from public.orders
  where code = p_order_code
    and payment_method = 'tarjeta';

  if not found then
    raise exception 'El pedido no existe o no admite pago con tarjeta.'
      using errcode = '22023';
  end if;

  if v_order.fulfillment_status = 'cancelled' then
    raise exception 'El pedido fue cancelado.' using errcode = '22023';
  end if;

  if v_order.payment_status = 'paid' then
    if v_order.payment_reference = v_reference then
      return jsonb_build_object(
        'code', v_order.code,
        'payment_status', v_order.payment_status,
        'paid_at', v_order.paid_at
      );
    end if;
    raise exception 'El pedido ya fue pagado con otra referencia.'
      using errcode = '22023';
  end if;

  update public.orders
  set payment_status = 'paid',
      paid_at = now(),
      payment_reference = v_reference
  where id = v_order.id
  returning * into v_order;

  return jsonb_build_object(
    'code', v_order.code,
    'payment_status', v_order.payment_status,
    'paid_at', v_order.paid_at
  );
end;
$$;

-- ---------------------------------------------------------------------------
-- admin_complete_order: B3 (idempotente)
-- ---------------------------------------------------------------------------
create or replace function public.admin_complete_order(p_order_code text)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Solo un administrador puede completar pedidos.'
      using errcode = '42501';
  end if;

  select * into v_order
  from public.orders
  where code = p_order_code;

  if not found then
    raise exception 'El pedido no existe.' using errcode = '22023';
  end if;

  if v_order.fulfillment_status = 'cancelled' then
    raise exception 'El pedido fue cancelado.' using errcode = '22023';
  end if;

  if v_order.fulfillment_status = 'completed' then
    return jsonb_build_object(
      'code', v_order.code,
      'payment_status', v_order.payment_status,
      'fulfillment_status', v_order.fulfillment_status,
      'completed_at', v_order.completed_at
    );
  end if;

  if v_order.payment_status <> 'paid' then
    raise exception 'El pedido debe estar pagado antes de completarse.'
      using errcode = '22023';
  end if;

  update public.orders
  set fulfillment_status = 'completed',
      completed_at = now()
  where id = v_order.id
  returning * into v_order;

  return jsonb_build_object(
    'code', v_order.code,
    'payment_status', v_order.payment_status,
    'fulfillment_status', v_order.fulfillment_status,
    'completed_at', v_order.completed_at
  );
end;
$$;

-- ---------------------------------------------------------------------------
-- admin_save_category: upsert de categoría + sync de category_products
-- ---------------------------------------------------------------------------
create or replace function public.admin_save_category(
  p_category_id uuid,
  p_name        text,
  p_description text,
  p_is_active   boolean,
  p_product_ids uuid[]
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_id uuid;
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Solo un administrador puede guardar categorias.'
      using errcode = '42501';
  end if;

  if p_name is null or length(trim(p_name)) = 0 then
    raise exception 'El nombre de la categoria es obligatorio.'
      using errcode = '22023';
  end if;

  if p_category_id is null then
    insert into public.categories (name, description, is_active)
    values (
      trim(p_name),
      coalesce(p_description, ''),
      coalesce(p_is_active, true)
    )
    returning id into v_id;
  else
    update public.categories
       set name        = trim(p_name),
           description = coalesce(p_description, ''),
           is_active   = coalesce(p_is_active, true)
     where id = p_category_id
     returning id into v_id;

    if v_id is null then
      raise exception 'La categoria no existe.' using errcode = 'P0002';
    end if;

    delete from public.category_products where category_id = v_id;
  end if;

  if p_product_ids is not null and array_length(p_product_ids, 1) > 0 then
    insert into public.category_products (category_id, product_id)
    select distinct v_id, prod_id
    from unnest(p_product_ids) as prod_id
    where prod_id is not null
    on conflict do nothing;
  end if;

  return v_id;
end;
$$;

-- ---------------------------------------------------------------------------
-- admin_save_product: upsert de producto + sync de category_products
-- ---------------------------------------------------------------------------
create or replace function public.admin_save_product(
  p_product_id   uuid,
  p_name         text,
  p_description  text,
  p_emoji        text,
  p_price        numeric,
  p_is_available boolean,
  p_category_ids uuid[]
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_id uuid;
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Solo un administrador puede guardar productos.'
      using errcode = '42501';
  end if;

  if p_name is null or length(trim(p_name)) = 0 then
    raise exception 'El nombre del producto es obligatorio.'
      using errcode = '22023';
  end if;

  if p_price is null or p_price < 0 then
    raise exception 'El precio debe ser mayor o igual a 0.'
      using errcode = '22023';
  end if;

  if p_product_id is null then
    insert into public.products (name, description, emoji, price, is_available)
    values (
      trim(p_name),
      coalesce(p_description, ''),
      coalesce(p_emoji, ''),
      p_price,
      coalesce(p_is_available, true)
    )
    returning id into v_id;
  else
    update public.products
       set name         = trim(p_name),
           description  = coalesce(p_description, ''),
           emoji        = coalesce(p_emoji, ''),
           price        = p_price,
           is_available = coalesce(p_is_available, true)
     where id = p_product_id
     returning id into v_id;

    if v_id is null then
      raise exception 'El producto no existe.' using errcode = 'P0002';
    end if;

    delete from public.category_products where product_id = v_id;
  end if;

  if p_category_ids is not null and array_length(p_category_ids, 1) > 0 then
    insert into public.category_products (category_id, product_id)
    select distinct cat_id, v_id
    from unnest(p_category_ids) as cat_id
    where cat_id is not null
    on conflict do nothing;
  end if;

  return v_id;
end;
$$;

-- ---------------------------------------------------------------------------
-- VISTAS DE REPORTES: B1 (agrupa cobros por paid_at) + B5 (tz Tijuana)
-- ---------------------------------------------------------------------------
create or replace view public.admin_sales_summary_by_day
with (security_invoker = true)
as
with paid as (
  select
    (orders.paid_at at time zone 'America/Tijuana')::date as sales_date,
    count(*)                                            as paid_orders,
    sum(orders.total_amount)::numeric(14, 2)            as collected_amount
  from public.orders as orders
  where orders.payment_status = 'paid'
    and public.is_admin()
  group by 1
),
pending as (
  select
    (orders.created_at at time zone 'America/Tijuana')::date as sales_date,
    count(*)                                                 as pending_orders,
    sum(orders.total_amount)::numeric(14, 2)                 as pending_amount
  from public.orders as orders
  where orders.payment_status = 'pending'
    and public.is_admin()
  group by 1
)
select
  coalesce(p.sales_date, d.sales_date)            as sales_date,
  coalesce(p.paid_orders, 0)                      as paid_orders,
  coalesce(p.collected_amount, 0::numeric(14, 2)) as collected_amount,
  coalesce(d.pending_orders, 0)                   as pending_orders,
  coalesce(d.pending_amount, 0::numeric(14, 2))   as pending_amount
from paid p
full outer join pending d on d.sales_date = p.sales_date;

create or replace view public.admin_top_products
with (security_invoker = true)
as
select
  items.product_id,
  items.product_name,
  sum(items.quantity)::bigint           as units_sold,
  sum(items.line_total)::numeric(14, 2) as sales_amount
from public.order_items as items
join public.orders as orders on orders.id = items.order_id
where orders.payment_status = 'paid'
  and public.is_admin()
group by items.product_id, items.product_name;

create policy profiles_update_self_or_admin
on public.profiles for update to authenticated
using (id = (select auth.uid()) or public.is_admin())
with check (id = (select auth.uid()) or public.is_admin());

create policy categories_read_active_or_admin
on public.categories for select to anon, authenticated
using (is_active or public.is_admin());

create policy categories_admin_manage
on public.categories for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy products_read_available_or_admin
on public.products for select to anon, authenticated
using (is_available or public.is_admin());

create policy products_admin_manage
on public.products for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy category_products_read_available_or_admin
on public.category_products for select to anon, authenticated
using (
  public.is_admin()
  or (
    exists (
      select 1 from public.categories as category
      where category.id = category_id and category.is_active
    )
    and exists (
      select 1 from public.products as product
      where product.id = product_id and product.is_available
    )
  )
);

create policy category_products_admin_manage
on public.category_products for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy orders_read_owner_or_admin
on public.orders for select to authenticated
using (customer_id = (select auth.uid()) or public.is_admin());

create policy order_items_read_owner_or_admin
on public.order_items for select to authenticated
using (
  exists (
    select 1 from public.orders as orders
    where orders.id = order_id
      and (orders.customer_id = (select auth.uid()) or public.is_admin())
  )
);

commit;
